"use client";

export type HapticStyle = "light" | "medium" | "heavy" | "success" | "warning" | "error";

/**
 * Triggers subtle hardware haptic feedback on supported mobile devices
 * while remaining completely safe on desktop browsers.
 */
export function triggerHaptic(style: HapticStyle = "light"): void {
  if (typeof window === "undefined" || !("vibrate" in navigator)) return;

  try {
    switch (style) {
      case "light":
        navigator.vibrate(8);
        break;
      case "medium":
        navigator.vibrate(18);
        break;
      case "heavy":
        navigator.vibrate(35);
        break;
      case "success":
        navigator.vibrate([12, 40, 20]);
        break;
      case "warning":
        navigator.vibrate([25, 50, 25]);
        break;
      case "error":
        navigator.vibrate([45, 40, 45, 40, 60]);
        break;
    }
  } catch {
    // Ignore on unsupported devices
  }
}

export function formatFriendlyAddress(address?: string | null, chars = 4): string {
  if (!address) return "";
  if (address.length <= chars * 2 + 2) return address;
  return `${address.slice(0, chars + 2)}…${address.slice(-chars)}`;
}

"use client";

import React, { useEffect, useState } from "react";
import { useSpring, useTransform } from "framer-motion";

interface AnimatedNumberProps {
  value: number;
  prefix?: string;
  suffix?: string;
  decimals?: number;
  className?: string;
}

export default function AnimatedNumber({
  value,
  prefix = "",
  suffix = "",
  decimals = 2,
  className = "",
}: AnimatedNumberProps) {
  const spring = useSpring(value, {
    mass: 0.6,
    stiffness: 110,
    damping: 18,
  });

  const display = useTransform(spring, (current) => {
    const safeVal = Number.isFinite(current) ? current : 0;
    return `${prefix}${safeVal.toLocaleString(undefined, {
      minimumFractionDigits: decimals,
      maximumFractionDigits: decimals,
    })}${suffix}`;
  });

  const [rendered, setRendered] = useState<string>(
    `${prefix}${value.toLocaleString(undefined, {
      minimumFractionDigits: decimals,
      maximumFractionDigits: decimals,
    })}${suffix}`
  );

  useEffect(() => {
    spring.set(value);
  }, [value, spring]);

  useEffect(() => {
    const unsubscribe = display.on("change", (latest) => {
      setRendered(latest);
    });
    return () => unsubscribe();
  }, [display]);

  return <span className={`tabular-nums ${className}`}>{rendered}</span>;
}

"use client";

import React, { useState, useMemo, useRef } from "react";
import { motion } from "framer-motion";
import { triggerHaptic } from "@/lib/ui/haptics";

export interface ChartPoint {
  timestamp: string;
  value: number;
}

interface InteractiveChartProps {
  baseValue: number;
  change24h: number;
  currencySymbol: string;
  timeframe: "24H" | "7D" | "1M";
  onTimeframeChange: (tf: "24H" | "7D" | "1M") => void;
  onScrubValue?: (point: ChartPoint | null) => void;
}

export default function InteractiveChart({
  baseValue,
  change24h,
  timeframe,
  onTimeframeChange,
  onScrubValue,
}: InteractiveChartProps) {
  const svgRef = useRef<SVGSVGElement | null>(null);
  const [hoverIndex, setHoverIndex] = useState<number | null>(null);

  const effectiveBase = baseValue > 0 ? baseValue : 2640.5;
  const isPositive = change24h >= 0;

  // Generate smooth, realistic portfolio curve anchored to current live value
  const points = useMemo<ChartPoint[]>(() => {
    const count = 32;
    const startRatio = 1 - change24h / 100;
    const startVal = effectiveBase * startRatio;
    const now = Date.now();
    const stepMs =
      timeframe === "24H"
        ? 45 * 60 * 1000
        : timeframe === "7D"
        ? 5 * 3600 * 1000
        : 22 * 3600 * 1000;

    return Array.from({ length: count }, (_, idx) => {
      const progress = idx / (count - 1);
      const wave =
        Math.sin(progress * Math.PI * (timeframe === "24H" ? 2.5 : 4)) *
          (effectiveBase * 0.018) +
        Math.cos(idx * 0.9) * (effectiveBase * 0.007);
      const interpolated =
        startVal + (effectiveBase - startVal) * progress + (idx === count - 1 ? 0 : wave);

      const date = new Date(now - (count - 1 - idx) * stepMs);
      const label =
        timeframe === "24H"
          ? date.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })
          : date.toLocaleDateString([], { month: "short", day: "numeric" });

      return {
        timestamp: label,
        value: Math.max(interpolated, 0.01),
      };
    });
  }, [effectiveBase, change24h, timeframe]);

  const { pathD, areaD, coords } = useMemo(() => {
    const width = 400;
    const height = 104;
    const padY = 12;

    const vals = points.map((p) => p.value);
    const min = Math.min(...vals);
    const max = Math.max(...vals);
    const range = max - min || 1;

    const mapped = points.map((pt, i) => {
      const x = (i / (points.length - 1)) * width;
      const y = height - padY - ((pt.value - min) / range) * (height - padY * 2);
      return { x, y, pt };
    });

    const d = mapped
      .map((c, i) => `${i === 0 ? "M" : "L"} ${c.x.toFixed(1)} ${c.y.toFixed(1)}`)
      .join(" ");

    const area = `${d} L ${width} ${height} L 0 ${height} Z`;
    return { pathD: d, areaD: area, coords: mapped };
  }, [points]);

  const handlePointerMove = (clientX: number) => {
    if (!svgRef.current) return;
    const rect = svgRef.current.getBoundingClientRect();
    const relX = Math.max(0, Math.min(clientX - rect.left, rect.width));
    const ratio = relX / rect.width;
    const idx = Math.round(ratio * (points.length - 1));

    if (idx !== hoverIndex) {
      setHoverIndex(idx);
      triggerHaptic("light");
      onScrubValue?.(points[idx]);
    }
  };

  const clearScrub = () => {
    setHoverIndex(null);
    onScrubValue?.(null);
  };

  const strokeColor = isPositive ? "#10B981" : "#F43F5E";
  const activeCoord = hoverIndex !== null ? coords[hoverIndex] : null;

  return (
    <div className="select-none">
      <div className="relative h-28 w-full mt-3">
        <svg
          ref={svgRef}
          viewBox="0 0 400 104"
          className="w-full h-full overflow-visible cursor-crosshair touch-none"
          onMouseMove={(e) => handlePointerMove(e.clientX)}
          onMouseLeave={clearScrub}
          onTouchMove={(e) => {
            if (e.touches[0]) handlePointerMove(e.touches[0].clientX);
          }}
          onTouchEnd={clearScrub}
        >
          <defs>
            <linearGradient id="interactiveChartGrad" x1="0" y1="0" x2="0" y2="1">
              <stop offset="0%" stopColor={strokeColor} stopOpacity="0.38" />
              <stop offset="100%" stopColor={strokeColor} stopOpacity="0.0" />
            </linearGradient>
          </defs>

          <path d={areaD} fill="url(#interactiveChartGrad)" />

          <motion.path
            key={timeframe}
            initial={{ pathLength: 0, opacity: 0.5 }}
            animate={{ pathLength: 1, opacity: 1 }}
            transition={{ duration: 0.75, ease: "easeOut" }}
            d={pathD}
            fill="none"
            stroke={strokeColor}
            strokeWidth="2.75"
            strokeLinecap="round"
            strokeLinejoin="round"
          />

          {activeCoord && (
            <>
              <line
                x1={activeCoord.x}
                y1={0}
                x2={activeCoord.x}
                y2={104}
                stroke="rgba(255,255,255,0.28)"
                strokeWidth="1.2"
                strokeDasharray="3 3"
              />
              <circle
                cx={activeCoord.x}
                cy={activeCoord.y}
                r="6"
                fill={strokeColor}
                stroke="#07090E"
                strokeWidth="2.5"
              />
            </>
          )}
        </svg>
      </div>

      {/* Timeframe Pills & Live Status */}
      <div className="flex items-center justify-between mt-2">
        <div className="flex gap-1.5 bg-slate-950/70 p-1 rounded-xl border border-white/5">
          {(["24H", "7D", "1M"] as const).map((tf) => {
            const active = timeframe === tf;
            return (
              <button
                key={tf}
                type="button"
                onClick={() => {
                  triggerHaptic("light");
                  onTimeframeChange(tf);
                }}
                className={`px-3 py-1 rounded-lg text-[11px] font-bold transition-all ${
                  active
                    ? "bg-indigo-600 text-white shadow-sm shadow-indigo-500/40"
                    : "text-slate-400 hover:text-white"
                }`}
              >
                {tf}
              </button>
            );
          })}
        </div>

        <div className="flex items-center gap-1.5 text-[11px] text-slate-400">
          <span className="w-2 h-2 rounded-full bg-emerald-400 animate-pulse" />
          <span>
            {activeCoord
              ? `${activeCoord.pt.timestamp}`
              : "Drag chart to inspect"}
          </span>
        </div>
      </div>
    </div>
  );
}

"use client";

import React, { createContext, useContext, useState, useCallback } from "react";
import { motion, AnimatePresence } from "framer-motion";
import {
  CheckCircle2,
  AlertTriangle,
  XCircle,
  Sparkles,
  X,
} from "lucide-react";
import { triggerHaptic } from "@/lib/ui/haptics";

export type ToastType = "success" | "warning" | "error" | "info";

export interface ToastItem {
  id: string;
  type: ToastType;
  title: string;
  description?: string;
}

interface ToastContextValue {
  pushToast: (toast: Omit<ToastItem, "id">) => void;
}

const ToastContext = createContext<ToastContextValue>({
  pushToast: () => {},
});

export function useToast() {
  return useContext(ToastContext);
}

export function ToastProvider({ children }: { children: React.ReactNode }) {
  const [toasts, setToasts] = useState<ToastItem[]>([]);

  const pushToast = useCallback((toast: Omit<ToastItem, "id">) => {
    const id = `${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
    if (toast.type === "success") triggerHaptic("success");
    else if (toast.type === "error") triggerHaptic("error");
    else if (toast.type === "warning") triggerHaptic("warning");
    else triggerHaptic("light");

    setToasts((prev) => [{ ...toast, id }, ...prev.slice(0, 2)]);
    setTimeout(() => {
      setToasts((prev) => prev.filter((t) => t.id !== id));
    }, 4200);
  }, []);

  const dismiss = (id: string) => {
    setToasts((prev) => prev.filter((t) => t.id !== id));
  };

  return (
    <ToastContext.Provider value={{ pushToast }}>
      {children}
      <div
        aria-live="polite"
        className="fixed top-4 left-1/2 -translate-x-1/2 w-[calc(100%-2rem)] max-w-md z-[60] pointer-events-none space-y-2"
      >
        <AnimatePresence>
          {toasts.map((t) => (
            <motion.div
              key={t.id}
              initial={{ opacity: 0, y: -20, scale: 0.95 }}
              animate={{ opacity: 1, y: 0, scale: 1 }}
              exit={{ opacity: 0, y: -15, scale: 0.95 }}
              className="pointer-events-auto p-3.5 rounded-2xl bg-slate-900/95 border border-white/15 backdrop-blur-xl shadow-2xl flex items-start gap-3 text-white"
            >
              {t.type === "success" && (
                <CheckCircle2 className="w-5 h-5 text-emerald-400 shrink-0 mt-0.5" />
              )}
              {t.type === "warning" && (
                <AlertTriangle className="w-5 h-5 text-amber-400 shrink-0 mt-0.5" />
              )}
              {t.type === "error" && (
                <XCircle className="w-5 h-5 text-rose-400 shrink-0 mt-0.5" />
              )}
              {t.type === "info" && (
                <Sparkles className="w-5 h-5 text-indigo-400 shrink-0 mt-0.5" />
              )}

              <div className="flex-1 min-w-0">
                <div className="text-xs font-bold">{t.title}</div>
                {t.description && (
                  <div className="text-[11px] text-slate-300 mt-0.5 break-words">
                    {t.description}
                  </div>
                )}
              </div>

              <button
                onClick={() => dismiss(t.id)}
                className="text-slate-400 hover:text-white p-1"
              >
                <X className="w-3.5 h-3.5" />
              </button>
            </motion.div>
          ))}
        </AnimatePresence>
      </div>
    </ToastContext.Provider>
  );
}

export default function SwapView() {
  const {
    activeAddress,
    activeChainId,
    customTokens,
    sessionPrivateKey,
    addTransaction,
  } = useWalletStore();
  const { pushToast } = useToast();

  const chainTokens = useMemo(
    () => customTokens.filter((t) => t.chainId === activeChainId),
    [customTokens, activeChainId]
  );

  const [sellIdx, setSellIdx] = useState<number>(0);
  const [buyIdx, setBuyIdx] = useState<number>(1);
  const [sellAmount, setSellAmount] = useState<string>("0.1");
  const [slippage, setSlippage] = useState<number>(0.5);
  const [quote, setQuote] = useState<SwapQuoteResponse | null>(null);
  const [securityAudit, setSecurityAudit] =
    useState<SecurityAuditResult | null>(null);
  const [needsApproval, setNeedsApproval] = useState<boolean>(false);
  const [unlimitedApproval, setUnlimitedApproval] = useState<boolean>(false);
  const [isLoadingQuote, setIsLoadingQuote] = useState<boolean>(false);
  const [showPreview, setShowPreview] = useState<boolean>(false);
  const [tokenPickerSide, setTokenPickerSide] = useState<
    "sell" | "buy" | null
  >(null);
  const [tokenSearch, setTokenSearch] = useState<string>("");
  const [txError, setTxError] = useState<string | null>(null);
  const [txSuccessHash, setTxSuccessHash] = useState<string | null>(null);
  const [countdown, setCountdown] = useState<number>(10);

  const sellToken = chainTokens[sellIdx] || chainTokens[0];
  const buyToken = chainTokens[buyIdx] || chainTokens[1] || chainTokens[0];

  const loadQuoteAndChecks = async () => {
    if (!sellToken || !buyToken || !sellAmount || parseFloat(sellAmount) <= 0)
      return;
    setIsLoadingQuote(true);
    setTxError(null);
    try {
      const [q, audit] = await Promise.all([
        fetchSwapQuote({
          chainId: activeChainId,
          sellToken,
          buyToken,
          sellAmountHuman: sellAmount,
          slippagePercent: slippage,
          takerAddress: activeAddress || undefined,
        }),
        auditTokenSecurity(activeChainId, buyToken.address),
      ]);
      setQuote(q);
      setSecurityAudit(audit);
      setCountdown(10);

      if (sellToken.address !== "NATIVE" && activeAddress) {
        const currentAllowance = await checkTokenAllowance({
          chainId: activeChainId,
          tokenAddress: sellToken.address,
          owner: activeAddress,
          spender: q.allowanceTarget,
        }).catch(() => 0n);

        const required = parseUnits(sellAmount, sellToken.decimals);
        setNeedsApproval(currentAllowance < required);
      } else {
        setNeedsApproval(false);
      }
    } catch (err: unknown) {
      setTxError(err instanceof Error ? err.message : "Failed to fetch quote");
    } finally {
      setIsLoadingQuote(false);
    }
  };

  useEffect(() => {
    loadQuoteAndChecks();
    const timer = setInterval(() => {
      setCountdown((c) => {
        if (c <= 1) {
          loadQuoteAndChecks();
          return 10;
        }
        return c - 1;
      });
    }, 1000);
    return () => clearInterval(timer);
  }, [sellIdx, buyIdx, sellAmount, slippage, activeChainId]);

    if (!sessionPrivateKey || !quote || sellToken.address === "NATIVE") return;
    setTxError(null);
    triggerHaptic("medium");
    try {
      const amountToApprove = unlimitedApproval
        ? maxUint256
        : parseUnits(sellAmount, sellToken.decimals);

      const data = encodeErc20Approve(quote.allowanceTarget, amountToApprove);
      const sim = await simulateAndEstimateTransaction({
        chainId: activeChainId,
        from: activeAddress!,
        to: sellToken.address,
        data,
      });

      const tx = await signAndBroadcastTransaction({
        chainId: activeChainId,
        privateKeyHex: sessionPrivateKey,
        to: sellToken.address,
        value: 0n,
        data,
        gasLimit: sim.fees.estimatedGasLimit,
        maxFeePerGas: sim.fees.normal.maxFeePerGas,
        maxPriorityFeePerGas: sim.fees.normal.maxPriorityFeePerGas,
        type: "approve",
        summary: `Approve ${sellToken.symbol} for Swap`,
      });

      addTransaction(tx);

        type: "success",
        title: `${sellToken.symbol} Approved`,
        description: "Exact token allowance confirmed. Ready to swap.",
      });
    } catch (err: unknown) {
      setTxError(err instanceof Error ? err.message : "Approval failed");
    }
  };

      setTxError("Unlock your wallet to sign swap transactions.");
      return;
    }
    setTxError(null);
    triggerHaptic("medium");
    try {

        chainId: activeChainId,
        from: activeAddress,
        to: quote.to,
        value: BigInt(quote.value),
        data: quote.data,
      });

        pushToast({
          type: "error",
          title: "Simulation Protected Your Funds",
          description: sim.revertReason || "Transaction would revert on-chain.",

        title: "Swap Broadcasted!",
        description: `Swapping ${sellAmount} ${sellToken.symbol} for ${buyToken.symbol}`,
      });
    } catch (err: unknown) {

      )

================================================================================
"use client";

import React, { useState, useEffect } from "react";
import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { useWalletStore } from "@/store/useWalletStore";
import { ToastProvider } from "@/components/ui/ToastProvider";

export default function Providers({ children }: { children: React.ReactNode }) {
  const [queryClient] = useState(
    () =>
      new QueryClient({
        defaultOptions: {
          queries: {
            staleTime: 15_000,
            retry: 2,
          },
        },
      })
  );

  const {
    isUnlocked,
    lastActiveTimestamp,
    autoLockMinutes,
    lockWallet,
    touchActivity,
  } = useWalletStore();

  useEffect(() => {
    if (!isUnlocked) return;
    const interval = setInterval(() => {
      const elapsedMs = Date.now() - lastActiveTimestamp;
      if (elapsedMs > autoLockMinutes * 60 * 1000) {
        lockWallet();
      }
    }, 10_000);

    const handleUserActivity = () => touchActivity();
    window.addEventListener("pointerdown", handleUserActivity);
    window.addEventListener("keydown", handleUserActivity);

    return () => {
      clearInterval(interval);
      window.removeEventListener("pointerdown", handleUserActivity);
      window.removeEventListener("keydown", handleUserActivity);
    };
  }, [isUnlocked, lastActiveTimestamp, autoLockMinutes]);

  return (
    <QueryClientProvider client={queryClient}>
      <ToastProvider>{children}</ToastProvider>
    </QueryClientProvider>
  );
}

              key={tab}
              initial={{ opacity: 0, y: 10 }}
              animate={{ opacity: 1, y: 0 }}
              exit={{ opacity: 0, y: -10 }}
              transition={{ duration: 0.18, ease: "easeOut" }}
